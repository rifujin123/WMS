namespace WMS.Domain.Entities;

public class Product
{
    public Guid Id { get; private set; }
    public string Code { get; private set; }
    public string Name { get; private set; }
    public string? Description { get; private set; }
    public ProductUnit Unit { get; private set; }
    public decimal Price { get; private set; }
    public int StockQuantity { get; private set; }
    public bool IsActive { get; private set; }

    private Product() { }  // EF Core cần constructor không tham số

    public Product(string code, string name, ProductUnit unit, decimal price)
    {
        Id = Guid.NewGuid();
        Code = code; Name = name; Unit = unit; Price = price;
        StockQuantity = 0; IsActive = true;
    }

    // Business rules NGAY trong Entity
    public void UpdatePrice(decimal newPrice)
    {
        if (newPrice < 0)
            throw new ArgumentException("Price must be >= 0");
        Price = newPrice;
    }

    public void AdjustStock(int quantity)
    {
        if (StockQuantity + quantity < 0)
            throw new InvalidOperationException("Insufficient stock");
        StockQuantity += quantity;
    }

    public void Deactivate() => IsActive = false;
}
✅ Domain chỉ chứa: business rules, validation. Không biết EF Core, DTO, hay Controller.

src/WMS.Application/Services/ProductService.cs

using WMS.Application.DTOs;
using WMS.Application.Interfaces;

namespace WMS.Application.Services;

public class ProductService : IProductService
{
    private readonly IProductRepository _repository;

    public ProductService(IProductRepository repository)
        => _repository = repository;

    public async Task<IEnumerable<ProductDto>> GetAllAsync()
    {
        var products = await _repository.GetAllAsync();
        return products.Select(MapToDto);
    }

    public async Task<ProductDto?> GetByIdAsync(Guid id)
    {
        var product = await _repository.GetByIdAsync(id);
        return product is null ? null : MapToDto(product);
    }

    public async Task<ProductDto> CreateAsync(CreateProductRequest request)
    {
        var unit = Enum.Parse<Domain.Enums.ProductUnit>(request.Unit);
        var product = new Domain.Entities.Product(
            request.Code, request.Name, unit, request.Price
        );
        await _repository.AddAsync(product);
        return MapToDto(product);
    }

    public async Task DeleteAsync(Guid id)
        => await _repository.DeleteAsync(id);

    private static ProductDto MapToDto(Domain.Entities.Product product)
        => new(product.Id, product.Code, product.Name,
              product.Description, product.Unit.ToString(),
              product.Price, product.StockQuantity, product.IsActive);
}
✅ Application chỉ: orchestrate use case, map Entity ↔ DTO, gọi interface. Không biết EF Core, HttpContext, hay database.

4️⃣ Bước 3: Infrastructure — DbContext + Repository

Nơi đặt: src/WMS.Infrastructure/

Infrastructure

src/WMS.Infrastructure/Data/AppDbContext.cs

using Microsoft.EntityFrameworkCore;
using WMS.Domain.Entities;

namespace WMS.Infrastructure.Data;

public class AppDbContext : DbContext
{
    public DbSet<Product> Products => Set<Product>();

    public AppDbContext(DbContextOptions<AppDbContext> options) : base(options) { }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<Product>(entity =>
        {
            entity.HasKey(e => e.Id);
            entity.Property(e => e.Code).HasMaxLength(50).IsRequired();
            entity.Property(e => e.Name).HasMaxLength(200).IsRequired();
            entity.Property(e => e.Price).HasColumnType("decimal(18,2)");
            entity.HasIndex(e => e.Code).IsUnique();
        });
    }
}
Infrastructure

src/WMS.Infrastructure/Repositories/ProductRepository.cs

using Microsoft.EntityFrameworkCore;
using WMS.Application.Interfaces;
using WMS.Domain.Entities;
using WMS.Infrastructure.Data;

namespace WMS.Infrastructure.Repositories;

public class ProductRepository : IProductRepository
{
    private readonly AppDbContext _context;

    public ProductRepository(AppDbContext context) => _context = context;

    public async Task<Product?> GetByIdAsync(Guid id)
        => await _context.Products.FindAsync(id);

    public async Task<IEnumerable<Product>> GetAllAsync()
        => await _context.Products.Where(p => p.IsActive).ToListAsync();

    public async Task AddAsync(Product product)
    {
        await _context.Products.AddAsync(product);
        await _context.SaveChangesAsync();
    }

    public async Task UpdateAsync(Product product)
    {
        _context.Products.Update(product);
        await _context.SaveChangesAsync();
    }

    public async Task DeleteAsync(Guid id)
    {
        var product = await _context.Products.FindAsync(id);
        if (product is not null)
        {
            _context.Products.Remove(product);
            await _context.SaveChangesAsync();
        }
    }
}

5️⃣ Bước 4: API — Controller + DI Registration

Nơi đặt: src/WMS.API/

API

src/WMS.API/Controllers/ProductsController.cs

using Microsoft.AspNetCore.Mvc;
using WMS.Application.DTOs;
using WMS.Application.Services;

namespace WMS.API.Controllers;

[ApiController]
[Route("api/products")]
public class ProductsController : ControllerBase
{
    private readonly IProductService _service;

    public ProductsController(IProductService service) => _service = service;

    [HttpGet]
    public async Task<IActionResult> GetAll()
    {
        var products = await _service.GetAllAsync();
        return Ok(products);
    }

    [HttpGet("{id}")]
    public async Task<IActionResult> GetById(Guid id)
    {
        var product = await _service.GetByIdAsync(id);
        return product is null ? NotFound() : Ok(product);
    }

    [HttpPost]
    public async Task<IActionResult> Create([FromBody] CreateProductRequest request)
    {
        var product = await _service.CreateAsync(request);
        return CreatedAtAction(nameof(GetById), new { id = product.Id }, product);
    }

    [HttpDelete("{id}")]
    public async Task<IActionResult> Delete(Guid id)
    {
        await _service.DeleteAsync(id);
        return NoContent();
    }
}
API

src/WMS.API/Extensions/ServiceCollectionExtensions.cs

using WMS.Application.Interfaces;
using WMS.Application.Services;
using WMS.Infrastructure.Data;
using WMS.Infrastructure.Repositories;

namespace WMS.API.Extensions;

public static class ServiceCollectionExtensions
{
    public static IServiceCollection AddApplicationServices(
        this IServiceCollection services,
        string connectionString)
    {
        services.AddDbContext<AppDbContext>(options =>
            options.UseSqlServer(connectionString));

        services.AddScoped<IProductRepository, ProductRepository>();
        services.AddScoped<IProductService, ProductService>();

        return services;
    }
}

6️⃣ Bước 5: Migration + NuGet Packages

Cài package

# Infrastructure
dotnet add src/WMS.Infrastructure/WMS.Infrastructure.csproj package Microsoft.EntityFrameworkCore.SqlServer
dotnet add src/WMS.Infrastructure/WMS.Infrastructure.csproj package Microsoft.EntityFrameworkCore.Tools

# API (cần EF Core Design để chạy migration)
dotnet add src/WMS.API/WMS.API.csproj package Microsoft.EntityFrameworkCore.Design

Migration & Database

# Tạo migration
dotnet ef migrations add InitialCreate \
    --project src/WMS.Infrastructure \
    --startup-project src/WMS.API

# Cập nhật database
dotnet ef database update \
    --project src/WMS.Infrastructure \
    --startup-project src/WMS.API

# Chạy API
dotnet run --project src/WMS.API

Flow request hoàn chỉnh — Quá trình 1 request chạy qua các layer

HTTP POST /api/products
  Body: { "code": "P001", "name": "Product 1", "unit": "Piece", "price": 1000 }
  │
  ▼
[1] ProductsController.Create(CreateProductRequest)         ← API
  │
  ▼
[2] ProductService.CreateAsync(request)                      ← Application
  │   ├── parse "Piece" → ProductUnit.Piece
  │   ├── new Product("P001", "Product 1", Piece, 1000)         ← Domain
  │   └── _repository.AddAsync(product)
  │
  ▼
[3] ProductRepository.AddAsync(product)                      ← Infrastructure
  │   ├── _context.Products.AddAsync(product)
  │   └── _context.SaveChangesAsync()
  │
  ▼
[4] ProductService.MapToDto(product)                         ← Application
  │   └── return ProductDto
  │
  ▼
[5] Controller → CreatedAtAction(dto)                        ← API
  │
  ▼
HTTP 201 Created
  Body: { "id": "...", "code": "P001", "name": "Product 1", ... }

7️⃣ Quy tắc vàng

Layer File Được làm ✅ Không được làm ❌
Domain Entities/* Business rules, validation, constructor Dùng EF Core, biết DTO, biết Controller
Domain Enums/* Khai báo hằng số —
Application Interfaces/* Khai báo method Có implementation
Application Services/* Gọi interface, map Entity↔DTO, orchestrate Gọi DbContext, biết HttpContext
Application DTOs/* Data contract, validation attributes Business logic
Infrastructure Data/* DbContext, EF config, Fluent API Business rules
Infrastructure Repositories/* EF Core, SQL queries Logic nghiệp vụ
API Controllers/* Http → Service → Response Gọi DB trực tiếp

8️⃣ Checklist — Khi tạo entity mới

  • ☐ Domain: Tạo Entity + Enum (nếu cần)
  • ☐ Domain: Set private set; cho properties, private Product() { } cho EF
  • ☐ Domain: Viết business method trong Entity (không để Service xử lý)
  • ☐ Application: Tạo Repository Interface
  • ☐ Application: Tạo DTO (Request + Response)
  • ☐ Application: Tạo Service Interface + Implementation
  • ☐ Infrastructure: Config Entity trong DbContext (Fluent API)
  • ☐ Infrastructure: Implement Repository
  • ☐ API: Tạo Controller
  • ☐ API: Đăng ký DI trong ServiceCollectionExtensions.cs
  • ☐ Chạy: dotnet ef migrations add + dotnet ef database update
